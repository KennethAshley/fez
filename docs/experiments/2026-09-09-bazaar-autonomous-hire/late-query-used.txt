import fs from "node:fs";
import assert from "node:assert/strict";
import { createRequire } from "node:module";
const require = createRequire("/Users/ken/Projects/Fez/fez-bazaar/package.json");
const WS = require("ws");
const { verifyEvent } = require("nostr-tools");
const folder = "/Users/ken/Projects/Fez/fez/docs/experiments/2026-09-09-bazaar-autonomous-hire";
const taskId = "bc1c5dd3618c2a8fd6117a5f04c38f10cfcefd3ac0d88a4ac0bd513ad8066830";
const target = "b7f0b1f5dc205d8c5ab07db0c7e61c20957d845c52bf21ee377469747430792f";
const events: { kind: number; created_at: number; content: string }[] = [];
await new Promise<void>((resolve,reject) => {
  const ws = new WS("wss://bazaar.fez.chat");
  const timer = setTimeout(() => { ws.close(); reject(new Error("read timed out")); }, 10000);
  ws.on("open", () => ws.send(JSON.stringify(["REQ","late",{kinds:[47002,47003],authors:[target],"#e":[taskId]}])));
  ws.on("error", (e: Error) => { clearTimeout(timer); reject(e); });
  ws.on("message", (raw: unknown) => {
    const msg = JSON.parse(String(raw));
    if (msg[0] === "EVENT") { assert(verifyEvent(msg[2])); events.push(msg[2]); }
    if (msg[0] === "EOSE") { clearTimeout(timer); ws.close(); resolve(); }
  });
});
const wire = JSON.parse(fs.readFileSync(`${folder}/wire.json`,"utf8"));
const task = wire.find((e: { id: string }) => e.id === taskId);
const result = events.find(e => e.kind === 47003);
assert(result, "no terminal reply yet");
const answer = JSON.parse(result.content);
fs.writeFileSync(`${folder}/late-reply.json`, JSON.stringify({at:new Date().toISOString(),taskId,target,receivedAfterBuyerFinished:true,delaySeconds:result.created_at-task.created_at,events},null,2)+"\n");
console.log(JSON.stringify({delaySeconds:result.created_at-task.created_at,status:answer.status,preview:String(answer.result).slice(0,280)}));
