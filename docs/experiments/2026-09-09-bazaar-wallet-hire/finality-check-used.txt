import assert from "node:assert/strict";
import fs from "node:fs";
import { loadConfig } from "/Users/ken/Projects/Fez/fez/packages/fez-wallet/src/config.ts";
import { subtensorFor } from "/Users/ken/Projects/Fez/fez/packages/fez-wallet/src/stake.ts";
const folder = "/Users/ken/Projects/Fez/fez/docs/experiments/2026-09-09-bazaar-wallet-hire";
const verified = JSON.parse(fs.readFileSync(`${folder}/verification.json`, "utf8"));
const api = await subtensorFor(loadConfig().endpoints.tao);
try {
  const paidHeader = await api.rpc.chain.getHeader(verified.blockRef);
  const finalHash = await api.rpc.chain.getFinalizedHead();
  const finalHeader = await api.rpc.chain.getHeader(finalHash);
  assert(finalHeader.number.toBigInt() >= paidHeader.number.toBigInt());
  assert.equal((await api.rpc.chain.getBlockHash(paidHeader.number)).toHex(), verified.blockRef);
  const result = { at: new Date().toISOString(), blockRef: verified.blockRef, blockNumber: paidHeader.number.toString(), finalizedBlock: finalHeader.number.toString(), finalized: true };
  fs.writeFileSync(`${folder}/finality.json`, JSON.stringify(result, null, 2) + "\n");
  console.log(JSON.stringify(result));
} finally { await api.disconnect(); }
